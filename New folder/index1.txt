  <script>
    const getStartedButton = document.querySelector('button');
    const venueInput = document.querySelector('input');

    getStartedButton.addEventListener('click', async () => {
      const userTypedValue = venueInput.value;

      if (userTypedValue.trim() === "") {
        alert("Please type a venue name first!");
        return;
      }

      // Pack the user input into a JavaScript Object layout
      const payloadData = {
        venueName: userTypedValue
      };

      try {
        // Shoot the data package across your network port to your Node server
        const response = await fetch('http://localhost:5000/api/get-started', {
          method: 'POST',
          headers: {
            'Content-Type': 'application/json'
          },
          body: JSON.stringify(payloadData) // Turn our object package into text strings
        });

        // Catch the JSON response response message coming back from the server
        const serverResult = await response.json();

        // Pop up the response text message that our server created for us!
        alert("Response Received From Server:\n" + serverResult.message);
        
        // Clear the text box layout clean
        venueInput.value = "";

      } catch (error) {
        alert("Network Error: Could not reach the server. Make sure node server.js is running!");
      }
    });

    // A. Query and build the live show schedule grid view
    async function fetchLiveSchedule() {
      try {
        const res = await fetch('http://localhost:5000/api/schedule');
        const shows = await res.json();
        const container = document.getElementById('scheduleGrid');
        
        container.innerHTML = shows.map(show => `
          <div style="background: #09090b; padding: 12px; border-radius: 8px; border-left: 4px solid #facc15;">
            <strong style="color: #ffffff;">${show.day}s:</strong> ${show.venue} 
            <div style="font-size: 12px; color: #71717a; margin-top: 4px;">${show.time} | ${show.address}</div>
          </div>
        `).join('');
      } catch (err) {
        document.getElementById('scheduleGrid').innerHTML = "<p style='color:red;'>Could not map itinerary updates.</p>";
      }
    }

    // B. Live filtering query engine for keyup inputs inside search bar
    const searchBar = document.getElementById('songSearchBar');
    searchBar.addEventListener('keyup', async () => {
      const currentQuery = searchBar.value;
      try {
        const res = await fetch(`http://localhost:5000/api/songs?q=${encodeURIComponent(currentQuery)}`);
        const matchingSongs = await res.json();
        const resultsBody = document.getElementById('songResultsBody');
        
        resultsBody.innerHTML = matchingSongs.map(song => `
          <tr>
            <td style="padding: 8px 0; border-bottom: 1px solid #27272a;">
              <div style="color: #ffffff; font-weight: bold; font-size:14px;">${song.title}</div>
              <div style="color: #a1a1aa; font-size: 12px;">${song.artist}</div>
            </td>
          </tr>
        `).join('');
      } catch (err) {
        console.error("Song filtering issue:", err);
      }
    });

    // Run layout configurations once window elements stabilize
    window.addEventListener('DOMContentLoaded', () => {
      fetchLiveSchedule();
      // Fire an initial empty search query to pre-load baseline target results
      searchBar.dispatchEvent(new Event('keyup'));
    });


  </script>
