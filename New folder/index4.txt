  <script>
    const submitBtn = document.getElementById('submitBtn');
    const venueInput = document.getElementById('venueName');
    submitBtn.addEventListener('click', async () => {
      const v = venueInput.value;
      if (!v.trim()) return alert("Enter a venue name.");
      try {
        const res = await fetch('http://localhost:5000/api/get-started', {
          method: 'POST',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify({ venueName: v })
        });
        const out = await res.json(); alert(out.message); venueInput.value = "";
      } catch (err) { alert("Make sure node server.js is running."); }
    });

    async function loadLiveSchedule() {
      try {
        const res = await fetch('http://localhost:5000/api/schedule');
        const list = await res.json();
        document.getElementById('scheduleGrid').innerHTML = list.map(s => `
          <div style="background:#050508; padding:14px; border-radius:8px; border-left:3px solid #ef4444; display:flex; justify-content:space-between; font-size:14px;">
            <strong>${s.day}s</strong><span style="color:#a1a1aa;">${s.venue}</span>
          </div>
        `).join('');
      } catch(e) {}
    }

    const searchBar = document.getElementById('songSearchBar');
    searchBar.addEventListener('keyup', async () => {
      try {
        const res = await fetch(`http://localhost:5000/api/songs?q=${encodeURIComponent(searchBar.value)}`);
        const songs = await res.json();
        document.getElementById('songResultsBody').innerHTML = songs.map(s => `
          <div style="padding:10px 0; border-bottom:1px solid #1c1c2b; display:flex; justify-content:space-between; font-size:14px;">
            <span style="color:white; font-weight:bold;">${s.title}</span><span style="color:#71717a;">${s.artist}</span>
          </div>
        `).join('');
      } catch(e) {}
    });

    window.addEventListener('DOMContentLoaded', () => {
      loadLiveSchedule();
      searchBar.dispatchEvent(new Event('keyup'));
    });
  </script>
