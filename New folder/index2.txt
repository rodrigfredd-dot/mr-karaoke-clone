  <script>
    // Bind network interaction endpoints directly
    const submitBtn = document.getElementById('submitBtn');
    const venueInput = document.getElementById('venueName');

    submitBtn.addEventListener('click', async () => {
      const userValue = venueInput.value;
      if (!userValue.trim()) return alert("Enter a venue name.");

      try {
        const res = await fetch('http://localhost:5000/api/get-started', {
          method: 'POST',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify({ venueName: userValue })
        });
        const outcome = await res.json();
        alert(outcome.message);
        venueInput.value = "";
      } catch (err) {
        alert("Make sure node server.js is running.");
      }
    });

    // Handle show list rendering
    async function loadSchedule() {
      try {
        const res = await fetch('http://localhost:5000/api/schedule');
        const list = await res.json();
        document.getElementById('scheduleGrid').innerHTML = list.map(s => `
          <div style="background:#09090b; padding:10px; border-radius:8px; font-size:14px;">
            <strong style="color:#fff;">${s.day}:</strong> ${s.venue}
          </div>
        `).join('');
      } catch(e) {}
    }

    // Live search bindings
    const searchBar = document.getElementById('songSearchBar');
    searchBar.addEventListener('keyup', async () => {
      try {
        const res = await fetch(`http://localhost:5000/api/songs?q=${encodeURIComponent(searchBar.value)}`);
        const songs = await res.json();
        document.getElementById('songResultsBody').innerHTML = songs.map(s => `
          <div style="padding:6px 0; border-bottom:1px solid #27272a; color:#fff;">
            <strong>${s.title}</strong> - <span style="color:#a1a1aa;">${s.artist}</span>
          </div>
        `).join('');
      } catch(e) {}
    });

    window.addEventListener('DOMContentLoaded', () => {
      loadSchedule();
      searchBar.dispatchEvent(new Event('keyup'));
    });
  </script>
